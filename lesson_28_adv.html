Конструктор героя

средненовоеавтопроверка

Напиши функцию-конструктор Hero(name, hp) (или класс class Hero). У героя есть свойства name и hp и метод attack(target), который отнимает у цели 10 здоровья.

Показать ответ (сначала попробуй сам!)
function Hero(name, hp) {
    this.name = name
    this.hp = hp
    this.attack = function (target) {
        target.hp -= 10
    }
}

let arthur = new Hero('Артур', 100)
let goblin = new Hero('Гоблин', 30)
arthur.attack(goblin)
console.log(goblin.hp)
5

Класс Car

сложноновоеавтопроверка

Напиши класс Car:

  • в constructor(brand) сохраняются марка brand и скорость speed = 0
  • метод accelerate() увеличивает скорость на 10
  • метод brake() уменьшает скорость на 10, но не ниже 0
  • метод info() возвращает строку Лада: 20 км/ч

Показать ответ (сначала попробуй сам!)
class Car {
    constructor(brand) {
        this.brand = brand
        this.speed = 0
    }

    accelerate() {
        this.speed += 10
    }

    brake() {
        this.speed = Math.max(0, this.speed - 10)
    }

    info() {
        return this.brand + ': ' + this.speed + ' км/ч'
    }
}

let car = new Car('Лада')
car.accelerate()
car.accelerate()
console.log(car.info())
