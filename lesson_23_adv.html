Купи слона

легкоавтопроверка

Программа предлагает купить слона 🐘. Что бы ни ответил пользователь, она отвечает: Все говорят <ответ> а ты купи 🐘. Остановиться можно, только нажав «Отмена» (тогда prompt вернёт null). Проверку ответа вынеси в функцию buyElephant.

Показать ответ (сначала попробуй сам!)
function buyElephant(answer) {
    if (answer === null) {
        return false
    }
    return true
}

let userInput = prompt('Купи 🐘');
let answer = buyElephant(userInput);

while (answer) {
    userInput = prompt('Все говорят ' + userInput + ' а ты купи 🐘');
    answer = buyElephant(userInput);
}

Зеркальный текст

среднеавтопроверка

Напиши функцию reverseString(string), которая возвращает строку задом наперёд (с помощью цикла). Спроси текст у пользователя и выведи Результат: …

Показать ответ (сначала попробуй сам!)
let reverseString = function (string) {
    let tmp = ''
    for (let i = string.length; i > 0; i--) {
        tmp += string[i - 1]
    }
    return tmp
}

let userInput = prompt('Введите текст и я переверну его: ')

alert('Результат: ' + reverseString(userInput))

Вечеринка с пиццей

среднеавтопроверка

Напиши функцию pizzaCount(pizza, people), которая возвращает массив из двух строк: смайлики 😛 по числу гостей и 🍕 по числу пицц. Спроси, сколько будет людей и пицц, и выведи Будет людей - 😛😛😛 и пицц - 🍕🍕.

Показать ответ (сначала попробуй сам!)
let pizzaCount = function (pizza, people) {
    let party = ''
    let food = ''
    for (let i = 0; i < people; i++) {
        party += '😛'
    }
    for (let i = 0; i < pizza; i++) {
        food += '🍕'
    }
    return [party, food]
}

let peopleInput = Number(prompt('Сколько будет человек: '))
let pizzaInput = Number(prompt('Сколько будет пицц: '))

let people = pizzaCount(pizzaInput, peopleInput)[0]
let food = pizzaCount(pizzaInput, peopleInput)[1]

alert('Будет людей - '+ people + ' и пицц - ' + food)

Самое большое из трёх

легконовоеавтопроверка

Напиши функцию maxOfThree(a, b, c), которая возвращает самое большое из трёх чисел. Используй if, а не Math.max.

Показать ответ (сначала попробуй сам!)
function maxOfThree(a, b, c) {
    let max = a
    if (b > max) {
        max = b
    }
    if (c > max) {
        max = c
    }
    return max
}

console.log(maxOfThree(1, 5, 3))
